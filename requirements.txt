# zypherLL -- runtime dependencies.
#
#   pip install -r requirements.txt
#
# The model itself runs in llama.cpp's llama-server, which is not a Python
# package: it is downloaded on first start (see ZYPHER_LLAMA_DIR), along with
# the GGUF weights. Nothing below talks to the GPU, so there is no CUDA build
# of anything to get right.
#
# Verified against Python 3.11.9 on a 6 GB RTX 3060 Laptop GPU.
# Floors are set at the release that introduced something the code actually
# uses; upper bounds stop at the next major, where those APIs may move.
#
# Nothing here installs torch. The original torch + bitsandbytes engine
# (ZYPHER_BACKEND=transformers) needs requirements-transformers.txt as well.


# --- memory ---------------------------------------------------------------

# The vector index is a plain float32 matrix and a matmul, so numpy is the
# whole of it.
numpy>=1.24,<3

# The recall encoder, run on the CPU from its ONNX export (downloaded once,
# ~90 MB, into ZYPHER_EMBED_DIR). Same vectors as sentence-transformers gives
# for the same model, at about 160 MB of RAM instead of 800. If either package
# is missing, zypher/model/memory.py tries sentence-transformers and then falls
# back to hashed n-grams, which keeps the runner working but makes recall
# markedly worse -- paraphrases stop matching.
#
# 1.19 is the first onnxruntime built against numpy 2, which the range above
# allows. Verified with onnxruntime 1.23.2 and tokenizers 0.23.2.
onnxruntime>=1.19,<2
tokenizers>=0.15,<1


# --- live web lookup ------------------------------------------------------

# Imported lazily inside fetch_context, so its absence degrades to "answer
# from the weights" rather than a crash. The package was renamed from
# duckduckgo-search at 9.0, which is the name zypher/model/retrieval.py imports.
ddgs>=9.0,<10


# --- HTTP API (python main.py serve) ----------------------------------------

# Only needed to serve the model over HTTP; the console chat runs without them.
# 0.100 is the first FastAPI on pydantic 2, which the request models use.
fastapi>=0.100,<1
uvicorn>=0.23,<1
